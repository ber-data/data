-- Create a column to filter out non-JGI sequencing project records.
-- The following records (166,530 of 584,673 as of February 16, 2025) are considered to be JGI:
--
-- DOE Joint Genome Institute (JGI)
-- DOE Joint Genome Institute (JGI)|454 Life Sciences
-- DOE Joint Genome Institute (JGI)|Australian National University
-- DOE Joint Genome Institute (JGI)|GATC-Biotech AG, Konstanz, Germany
-- DOE Joint Genome Institute (JGI)|454 Life Sciences
-- DOE Joint Bioenergy Institute (JBEI)|DOE Joint Genome Institute (JGI)

-- PostgreSQL 11
ALTER TABLE gold_seqproj ADD COLUMN is_jgi_project boolean NOT NULL DEFAULT FALSE;
UPDATE gold_seqproj SET is_jgi_project = TRUE WHERE sequencing_centers ILIKE '%DOE Joint Genome Institute (JGI)%';

-- Create views for CSV exports and queries to support data export.

CREATE OR REPLACE VIEW jgi_geodata AS
SELECT s.gold_id AS gold_study_id,
    o.gold_id AS gold_organism_id,
    b.gold_id AS gold_biosample_id,
    p.gold_id AS gold_project_id,
    b.has_geodata AS samp_has_geodata,
    b.latitude AS samp_lat,
    b.longitude AS samp_lon,
    o.has_geodata AS org_has_geodata,
    o.latitude AS org_lat,
    o.longitude AS org_lon,
    p.its_proposal_id,
    p.its_sequencing_project_id,
    p.its_sample_id,
    p.pmo_project_id,
    p.ncbi_bioproject_accession,
    p.ncbi_biosample_accession,
    p.sra_experiment_ids,
    p.jgi_award_dois,
    p.pmid_genome_pub,
    p.project_funding,
    p.project_name,
    s.name,
    s.description
   FROM gold_seqproj p
     LEFT JOIN gold_biosample b ON p.biosample_gold_id::bpchar = b.gold_id
     LEFT JOIN gold_organism o ON p.organism_gold_id::bpchar = o.gold_id
     LEFT JOIN gold_study s ON p.study_gold_id::bpchar = s.gold_id
   WHERE p.is_jgi_project=TRUE;

-- Export geographic data for Map interface as CSV UTF-8

-- Export unique biosamples and organsims associated JGI sequencing projects and have geographic data.
\copy (SELECT DISTINCT(gold_biosample_id) AS gold_id,samp_lat AS latitude,samp_lon AS longitude FROM jgi_geodata WHERE samp_has_geodata=TRUE) To 'C:\temp\jgi_gold_biosample_geo.csv' WITH DELIMITER ',' CSV HEADER
-- Expect: 54,366
\copy (SELECT DISTINCT(gold_organism_id) AS gold_id,org_lat AS latitude,org_lon AS longitude FROM jgi_geodata WHERE org_has_geodata=TRUE) To 'C:\temp\jgi_gold_organism_geo.csv' WITH DELIMITER ',' CSV HEADER
-- Expect: 9,503
--
-- Alternatively, simply export the sequencing project IDs and prefer biosample locations over organism locations:
\copy (WITH combined AS (SELECT gold_project_id AS gold_id,samp_lat AS latitude,samp_lon AS longitude,1 AS priority FROM jgi_geodata WHERE samp_has_geodata = TRUE UNION ALL SELECT gold_project_id AS gold_id,org_lat AS latitude,org_lon AS longitude,2 AS priority FROM jgi_geodata WHERE org_has_geodata = TRUE), ranked AS (SELECT *, ROW_NUMBER() OVER (PARTITION BY gold_id ORDER BY priority) AS rn FROM combined) SELECT gold_id, latitude, longitude FROM ranked WHERE rn = 1) To 'C:\temp\jgi_seqproj_geo.csv' WITH DELIMITER ',' CSV HEADER;
-- Expect: 70,331
